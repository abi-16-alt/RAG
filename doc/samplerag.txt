### Minimal `rag_demo.py`
Here’s a ready-to-run starter file you can drop into `rag-demo`:

```python
import os
from dotenv import load_dotenv
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_pinecone import PineconeVectorStore
from langchain.chains import RetrievalQA
from pinecone import Pinecone
from langchain.schema import Document

# Load environment variables
load_dotenv()

# 1. Initialize Pinecone
pc = Pinecone(api_key=os.getenv("PINECONE_API_KEY"))
index_name = "rag-demo-index"
index = pc.Index(index_name)

# 2. Set up embeddings
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")

# 3. Connect LangChain to Pinecone
vectorstore = PineconeVectorStore(
    index=index,
    embedding=embeddings,
    text_key="text"
)

# 4. Insert sample documents
docs = [
    Document(page_content="Renewable energy reduces carbon emissions.", metadata={"source": "doc1"}),
    Document(page_content="Solar and wind are the fastest-growing energy sources.", metadata={"source": "doc2"}),
    Document(page_content="Hydropower is one of the oldest renewable technologies.", metadata={"source": "doc3"})
]
vectorstore.add_documents(docs)

# 5. Create retriever
retriever = vectorstore.as_retriever(search_kwargs={"k": 3})

# 6. Define the LLM
llm = ChatOpenAI(model="gpt-4", temperature=0)

# 7. Build RetrievalQA chain
qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    retriever=retriever,
    chain_type="stuff"
)

# 8. Run a query
query = "Which renewable sources are mentioned?"
result = qa_chain.run(query)
print(result)
```

   import os

   llm = ChatOpenAI(model="gpt-4", api_key=os.getenv("OPENAI_API_KEY"))
   qa_chain = RetrievalQA.from_chain_type(llm, retriever=retriever)

   query = "What medications are recommended for skin rashes?"
   answer = qa_chain.run(query)
   print(answer)
