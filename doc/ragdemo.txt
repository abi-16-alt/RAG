```python
import os
import pinecone
from sentence_transformers import SentenceTransformer
from transformers import pipeline
import PyPDF2

# --- PDF Loader ---
def extract_text_from_pdf(pdf_path):
    text = ""
    with open(pdf_path, "rb") as f:
        reader = PyPDF2.PdfReader(f)
        for page in reader.pages:
            page_text = page.extract_text()
            if page_text:
                text += page_text + "\n"
    return text

# --- Chunker ---
def chunk_text(text, chunk_size=200):
    words = text.split()
    chunks = []
    for i in range(0, len(words), chunk_size):
        chunk = " ".join(words[i:i+chunk_size])
        chunks.append(chunk)
    return chunks

# --- Pinecone Setup ---
api_key = os.getenv("PINECONE_API_KEY")
pinecone.init(api_key=api_key)

index_name = "pdf-rag-index"
embed_model = SentenceTransformer("all-MiniLM-L6-v2")

if index_name in pinecone.list_indexes():
    pinecone.delete_index(index_name)
    print(f"Deleted old index '{index_name}'")

pinecone.create_index(
    name=index_name,
    dimension=384,
    metric="cosine",
    pod_type="s1"
)

index = pinecone.Index(index_name)

# --- Insert PDF ---
pdf_text = extract_text_from_pdf(
    r"C:\s\chatbot\models\rag-demo\cdn\Skin_Rashes_Medication_Guide.pdf"
)
chunks = chunk_text(pdf_text)

vectors = []
for i, chunk in enumerate(chunks):
    section = "Medications" if "Medications:" in chunk else "Preventions" if "Preventions:" in chunk else "General"
    vectors.append({
        "id": f"pdf_chunk_{i}",
        "values": embed_model.encode(chunk).tolist(),
        "metadata": {"text": chunk, "section": section}
    })

index.upsert(vectors=vectors)
print(index.describe_index_stats())

# --- Generator Setup ---
generator = pipeline("text2text-generation", model="google/flan-t5-base")

# --- Agentic Retriever ---
def agentic_retrieve(query_text):
    if "medication" in query_text.lower():
        section = "Medications"
    elif "prevention" in query_text.lower():
        section = "Preventions"
    else:
        section = "General"

    query_emb = embed_model.encode(query_text).tolist()
    results = index.query(
        vector=query_emb,
        top_k=3,
        include_metadata=True,
        filter={"section": section}
    )
    return results

# --- Interactive Loop ---
print("\nPDF loaded into Pinecone. You can now ask questions about it!")
while True:
    query_text = input("\nAsk a question (or type 'exit' to quit): ")
    if query_text.lower() == "exit":
        print("Goodbye!")
        break

    try:
        results = agentic_retrieve(query_text)

        print("\nRetrieved chunks:")
        for match in results["matches"]:
            print("-", match["metadata"]["text"], "(score:", match["score"], ")")

        context = " ".join([m["metadata"]["text"] for m in results["matches"]])
        prompt = f"Based only on the context below, answer the question briefly.\n\nQuestion: {query_text}\nContext: {context}\nAnswer:"

        answer = generator(prompt, max_length=200, do_sample=False)[0]["generated_text"]
        print("\nFinal Answer:\n", answer)

    except Exception as e:
        print("Error:", e)
```
